cmake_minimum_required(VERSION 3.21)
project(cNEFX VERSION 0.2.0 LANGUAGES CXX)

# As a subproject, cNEFX builds no tests, installs nothing and keeps the
# consumer's runtime library; cnefx::cnefx exposes only include/.
if(PROJECT_IS_TOP_LEVEL)
  include(CTest)
endif()
include(GNUInstallDirs)
option(CNEFX_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(CNEFX_USE_SYSTEM_LIBRAW "Use system LibRaw instead of the pinned static build" OFF)
option(CNEFX_LTO "Enable release link-time optimization" ON)
option(CNEFX_STATIC_RUNTIME "Link the C++ runtime statically into a top-level build" ON)
option(CNEFX_INSTALL "Install and package the executable" ${PROJECT_IS_TOP_LEVEL})
set(CMAKE_CXX_EXTENSIONS OFF)
if(PROJECT_IS_TOP_LEVEL AND CNEFX_STATIC_RUNTIME)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(CNEFX_LTO)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT cnefx_ipo LANGUAGES CXX)
  if(cnefx_ipo)
    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
  endif()
endif()
set(THREADS_PREFER_PTHREAD_FLAG ON)
find_package(Threads REQUIRED)
include(cmake/LibRaw.cmake)

add_library(cnefx_core STATIC src/core.cpp src/sparse.cpp src/radial.cpp)
target_compile_features(cnefx_core PUBLIC cxx_std_20)
target_include_directories(cnefx_core PUBLIC include)
target_link_libraries(cnefx_core PRIVATE Threads::Threads)
set_target_properties(cnefx_core PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(cnefx_core PRIVATE /W4 /fp:strict)
else()
  target_compile_options(cnefx_core PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  if(CNEFX_SANITIZE)
    target_compile_options(cnefx_core PUBLIC -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
    target_link_options(cnefx_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()

add_library(cnefx_io STATIC src/io.cpp src/sha256.cpp src/tiff.cpp)
target_link_libraries(cnefx_io PUBLIC cnefx_core)
target_include_directories(cnefx_io PUBLIC src)
if(MSVC)
  target_compile_options(cnefx_io PRIVATE /W4)
else()
  target_compile_options(cnefx_io PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_library(cnefx_format STATIC src/raw.cpp src/metadata.cpp src/nefx.cpp)
target_link_libraries(cnefx_format PUBLIC cnefx_io PRIVATE cnefx_libraw)
target_compile_definitions(cnefx_format PRIVATE CNEFX_VERSION="${PROJECT_VERSION}")
if(MSVC)
  target_compile_options(cnefx_format PRIVATE /W4 /fp:strict)
else()
  target_compile_options(cnefx_format PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()

add_library(cnefx_merge STATIC src/discover.cpp src/merge.cpp src/report.cpp)
target_link_libraries(cnefx_merge PUBLIC cnefx_core PRIVATE cnefx_format Threads::Threads)
target_compile_definitions(cnefx_merge PRIVATE CNEFX_VERSION="${PROJECT_VERSION}")
add_library(cnefx::cnefx ALIAS cnefx_merge)
add_executable(cnefx src/cli.cpp)
target_link_libraries(cnefx PRIVATE cnefx_merge)
if(MSVC)
  target_compile_options(cnefx_merge PRIVATE /W4)
  target_compile_options(cnefx PRIVATE /W4)
else()
  target_compile_options(cnefx_merge PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(cnefx PRIVATE -Wall -Wextra -Wpedantic)
  if(MINGW)
    target_link_options(cnefx PRIVATE -municode)
  endif()
endif()

if(PROJECT_IS_TOP_LEVEL AND BUILD_TESTING)
  add_executable(cnefx_core_test tests/cpp/core_test.cpp)
  target_link_libraries(cnefx_core_test PRIVATE cnefx_core)
  target_include_directories(cnefx_core_test PRIVATE src)
  add_test(NAME core COMMAND cnefx_core_test)
  add_executable(cnefx_io_test tests/cpp/io_test.cpp)
  target_link_libraries(cnefx_io_test PRIVATE cnefx_io)
  add_test(NAME io COMMAND cnefx_io_test ${CMAKE_CURRENT_BINARY_DIR}/io-test)
  add_executable(cnefx_format_test tests/cpp/format_test.cpp)
  target_link_libraries(cnefx_format_test PRIVATE cnefx_format)
  add_test(NAME format COMMAND cnefx_format_test)
  add_executable(cnefx_merge_test tests/cpp/merge_test.cpp)
  target_link_libraries(cnefx_merge_test PRIVATE cnefx_merge cnefx_io)
  add_test(NAME merge COMMAND cnefx_merge_test)
endif()

if(PROJECT_IS_TOP_LEVEL AND CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CNEFX_STATIC_RUNTIME AND
   NOT CNEFX_SANITIZE)
  target_link_options(cnefx PRIVATE -static-libgcc -static-libstdc++)
endif()
if(APPLE)
  target_link_options(cnefx PRIVATE -Wl,-dead_strip)
elseif(UNIX)
  target_link_options(cnefx PRIVATE -Wl,--gc-sections)
elseif(MSVC)
  target_link_options(cnefx PRIVATE /OPT:REF /OPT:ICF)
endif()
if(NOT CNEFX_INSTALL)
  return()
endif()
# Build provenance is regenerated on every build, so a rebuild after a commit
# or edit records the current revision and working-tree state.
set(CNEFX_REVISION "@CNEFX_REVISION@")
set(CNEFX_DIRTY "@CNEFX_DIRTY@")
set(CNEFX_CONFIG "@CNEFX_CONFIG@")
configure_file(cmake/build-info.json.in build-info.json.in @ONLY)
add_custom_target(cnefx_build_info ALL
  COMMAND "${CMAKE_COMMAND}" "-DSOURCE=${PROJECT_SOURCE_DIR}"
    "-DINPUT=${CMAKE_CURRENT_BINARY_DIR}/build-info.json.in"
    "-DOUTPUT=${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/build-info.json" "-DCONFIG=$<CONFIG>"
    -P "${PROJECT_SOURCE_DIR}/cmake/BuildInfo.cmake"
  BYPRODUCTS "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/build-info.json"
  VERBATIM)
install(TARGETS cnefx RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/build-info.json"
  DESTINATION ${CMAKE_INSTALL_DATADIR}/cnefx)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/cnefx)
if(NOT CNEFX_USE_SYSTEM_LIBRAW)
  install(FILES "${cnefx_libraw_source_SOURCE_DIR}/COPYRIGHT"
    "${cnefx_libraw_source_SOURCE_DIR}/LICENSE.CDDL" "${cnefx_libraw_source_SOURCE_DIR}/LICENSE.LGPL"
    DESTINATION ${CMAKE_INSTALL_DATADIR}/cnefx/libraw)
  string(TOLOWER "${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}" cnefx_platform)
  set(CNEFX_PACKAGE_TARGET "${cnefx_platform}" CACHE STRING "Platform identifier for the archive")
  set(CPACK_PACKAGE_FILE_NAME "cnefx-${PROJECT_VERSION}-${CNEFX_PACKAGE_TARGET}")
  set(CPACK_PACKAGE_CHECKSUM SHA256)
  if(WIN32)
    set(CPACK_GENERATOR ZIP)
  else()
    set(CPACK_GENERATOR TGZ)
    set(CPACK_STRIP_FILES ON)
  endif()
  include(CPack)
endif()
