cmake_minimum_required(VERSION 3.20)
project(cNEFX VERSION 0.1.0 LANGUAGES CXX)

include(CTest)
option(CNEFX_SANITIZE "Enable address and undefined behavior sanitizers" OFF)

add_library(cnefx_core src/core.cpp src/sparse.cpp src/radial.cpp)
target_compile_features(cnefx_core PUBLIC cxx_std_20)
target_include_directories(cnefx_core PUBLIC include)
set_target_properties(cnefx_core PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(cnefx_core PRIVATE /W4 /fp:strict)
else()
  target_compile_options(cnefx_core PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  if(CNEFX_SANITIZE)
    target_compile_options(cnefx_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(cnefx_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()

add_library(cnefx_io src/io.cpp src/sha256.cpp src/tiff.cpp)
target_link_libraries(cnefx_io PUBLIC cnefx_core)
target_include_directories(cnefx_io PUBLIC src)
if(MSVC)
  target_compile_options(cnefx_io PRIVATE /W4)
else()
  target_compile_options(cnefx_io PRIVATE -Wall -Wextra -Wpedantic)
endif()

find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBRAW REQUIRED IMPORTED_TARGET libraw>=0.22)
add_library(cnefx_format src/raw.cpp src/metadata.cpp src/nefx.cpp)
target_link_libraries(cnefx_format PUBLIC cnefx_io PRIVATE PkgConfig::LIBRAW)
target_compile_definitions(cnefx_format PRIVATE CNEFX_VERSION="${PROJECT_VERSION}")
if(MSVC)
  target_compile_options(cnefx_format PRIVATE /W4 /fp:strict)
else()
  target_compile_options(cnefx_format PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()

add_library(cnefx_merge src/merge.cpp src/report.cpp)
target_link_libraries(cnefx_merge PUBLIC cnefx_format)
target_compile_definitions(cnefx_merge PRIVATE CNEFX_VERSION="${PROJECT_VERSION}")
add_library(cnefx::cnefx ALIAS cnefx_merge)
add_executable(cnefx src/cli.cpp)
target_link_libraries(cnefx PRIVATE cnefx_merge)
if(MSVC)
  target_compile_options(cnefx_merge PRIVATE /W4)
  target_compile_options(cnefx PRIVATE /W4)
else()
  target_compile_options(cnefx_merge PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(cnefx PRIVATE -Wall -Wextra -Wpedantic)
  if(MINGW)
    target_link_options(cnefx PRIVATE -municode)
  endif()
endif()

if(BUILD_TESTING)
  add_executable(cnefx_core_test tests/cpp/core_test.cpp)
  target_link_libraries(cnefx_core_test PRIVATE cnefx_core)
  add_test(NAME core COMMAND cnefx_core_test)
  add_executable(cnefx_io_test tests/cpp/io_test.cpp)
  target_link_libraries(cnefx_io_test PRIVATE cnefx_io)
  add_test(NAME io COMMAND cnefx_io_test ${CMAKE_CURRENT_BINARY_DIR}/io-test)
  add_executable(cnefx_format_test tests/cpp/format_test.cpp)
  target_link_libraries(cnefx_format_test PRIVATE cnefx_format)
  add_test(NAME format COMMAND cnefx_format_test)
  add_executable(cnefx_merge_test tests/cpp/merge_test.cpp)
  target_link_libraries(cnefx_merge_test PRIVATE cnefx_merge)
  add_test(NAME merge COMMAND cnefx_merge_test)
endif()
